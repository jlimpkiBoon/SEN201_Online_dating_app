Requirement
- Linux (Ubuntu 22.04 or newer recommended)
- Python 3.8 + venv module
- Basic terminal access


Requirement install 
sudo apt install python3 python3-venv python3-pip -y


Installation
# 1. Extract the installer archive
tar -xzf dating-app-installer.tar.gz
cd dating-app-installer


# 2. Ensure scripts are executable
chmod +x install.sh uninstall.sh


# 3. Run the installer
./install.sh


# 4. Launch the app
dating-app


Uninstallation
 cd dating-app-installer
./uninstall.sh
